% TABLE 3 : Waymo init spectrum  (the regime contrast)
% ---------------------------------------------------------------------
\begin{table}[t]
\centering
\caption{Waymo init-spectrum (SSE PSNR): SfM-init / MVS-init (densification
ceiling) / GS-Net-init. \textbf{Same dataset and cameras}, only frame density
changes: in the sparse wide-baseline regime the ceiling is large ($+2.40$) and
GS-Net is useful ($+0.64$); in the dense interpolation regime the ceiling is
tiny ($+0.40$) and GS-Net cannot help. A controlled proof that coverage
governs efficacy.}
\label{tab:spectrum}
\begin{tabular}{lcccccc}
\toprule
Regime & SfM & MVS & GS-Net & MVS$-$SfM & GS-Net$-$SfM \\
\midrule
Dense (interp., no holes)      & 27.40 & 27.80 & 26.41 & $+0.40$ & $-0.99$ \\
Sparse wide-baseline (holes)   & 18.03 & 20.44 & 18.68 & $\mathbf{+2.40}$ & $\mathbf{+0.64}$ \\
\bottomrule
\end{tabular}
\end{table}
